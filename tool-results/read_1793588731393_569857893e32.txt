   468→   468→      </div>
   469→   469→
   470→   470→      {/* Шапка: мини-голограмма, имя и настройки */}

   472→   472→        <div className="flex items-center gap-3">
   473→   473→          {/* Голограмма-орбис: тап — активировать микрофон, Дария слушает */}
   474→   474→          <button
   475→   475→            onClick={toggleMic}
   476→   476→            aria-label={micListening ? 'Остановить микрофон' : 'Активировать голосового помощника'}

   479→   479→            style={{ boxShadow: `0 0 26px -10px ${accent}` }}
   480→   480→          >

   482→   482→          </button>
   483→   483→          <div className="leading-tight">

   485→   485→            <p className="text-[11px] text-white/40">голосовой ИИ-ассистент</p>
   486→   486→          </div>
   487→   487→        </div>
   488→   488→
   489→   489→        <Button
   490→   490→          variant="outline"
   491→   491→          size="sm"
   492→   492→          onClick={() => setSettingsOpen(true)}
   493→   493→          className="gap-1.5 rounded-full border-white/15 bg-white/5 text-[11px] font-semibold uppercase tracking-wider text-white/80 hover:bg-white/10 hover:text-white"
   494→   494→        >
   495→   495→          <Settings className="size-4" />
   496→   496→          Настройки

   578→   578→          )}
   579→   579→
   580→   580→          {/* Строка ввода: микрофон · волна/текст · дополнительно · отправить */}

   586→   586→            <button

   593→   593→              )}

   607→   607→            {/* Волна во время записи — тап по ней тоже останавливает */}

   628→   628→                className="max-h-32 min-h-11 flex-1 resize-none border-0 bg-transparent px-2 py-2.5 text-sm text-white placeholder:text-white/30 focus-visible:ring-0 focus-visible:ring-offset-0"
   629→   629→              />

   647→   647→              disabled={!canSend}

   650→   650→                'size-11 shrink-0 rounded-full border transition-all disabled:opacity-30',
   651→   651→                canSend

   662→   662→          {/* Статус: состояние ассистента / автоскролл */}

   664→   664→            {statusLine}
   665→   665→          </p>
   666→   666→        </div>
   667→   667→      </main>
   668→   668→
   669→   669→      <MemoryDialog open={memoryOpen} onOpenChange={setMemoryOpen} onMemoryChanged={() => {}} />
   670→   670→      <SettingsDialog
   671→   671→        open={settingsOpen}
   672→   672→        onOpenChange={setSettingsOpen}
   673→   673→        voiceId={ttsVoice}
   674→   674→        onVoiceChange={setVoice}
   675→   675→        voiceOn={voiceOn}

   680→   680→          setSettingsOpen(false)

   683→   683→        onClearChat={() => {

   688→   688→    </div>
   689→   689→  )
   690→   690→}
