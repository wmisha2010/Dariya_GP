    46→ *  («найди в интернете музыку…», «сама включи…», «врубай рок») */
    47→const MUSIC_VERB_RE =
    48→  /(включи(?:те)?|включай(?:те)?|врубай(?:те)?|вруби(?:те)?|поставь(?:те)?|запусти(?:те)?|сыграй(?:те)?|играй(?:те)?|найди(?:те)?|поищи(?:те)?|открой(?:те)?|послушаем)/i

    50→const MUSIC_STOP_RE = /(выключи(?:те)?|выруб[а-яё]*|останови(?:те)?|заглуши[а-яё]*|хватит)/i
    51→/** Отрицание: «не включай музыку» — это не команда играть */
    52→const MUSIC_NEGATION_RE = /не\s+(?:включай|играй|запускай|ставь|трогай)/i

    58→/** «дальше / следующий трек» — перелистать очередь */
    59→const MUSIC_NEXT_RE =

    61→
    62→/** Мусор в команде: «включи музыку из интернета» — ищем без «из интернета» */
    63→const MUSIC_FILLER_RE =
    64→  /\s*(?:из\s+(?:интернета?|сети|веба)|в\s+интернете?|в\s+онлайне?|онлайн|пожалуйста|что-?нибудь|что\s+нибудь|любую|любим[а-яё]*|какую-?нибудь|сама(?![а-яё])|сами(?![а-яё])|мне)\s*/gi
    65→/** Общие слова в начале запроса: «музыку Кино» → «Кино», «лирику Сектор Газа» → «Сектор Газа» */
    66→const MUSIC_NOUN_HEAD_RE =
    67→  /^\s*(?:музыку|музыка|музыки|песню|песня|песенку|песни|трек|трека|треки|композицию|композиция|клип|плейлист|ремикс|альбом|сборник|хит|мелодию|мелодия|лирику|лирика|лирики|лирик[а-яё]*)\s+/i
    68→/** Запрос состоит только из общих слов — конкретики нет, играем популярное */
    69→const MUSIC_NOUN_ONLY_RE =
    70→  /^\s*(?:музык[а-яё]*|песн[а-яё]*|трек[а-яё]*|клип[а-яё]*|плейлист[а-яё]*|ремикс[а-яё]*|альбом[а-яё]*|сборник[а-яё]*|хит[а-яё]*|мелоди[а-яё]*|лирик[а-яё]*|чего-?нибудь|что-?нибудь)\s*$/i

    81→/** «найди в интернете музыку Сектор Газа, пожалуйста» → «сектор газа», «включи лирику» → '' (популярное) */
    82→function extractMusicQuery(raw: string): string {
    83→  let q = raw.replace(MUSIC_VERB_RE, ' ').replace(MUSIC_FILLER_RE, ' ')
    84→  let prev = ''
    85→  while (prev !== q) {
    86→    prev = q
    87→    q = q.replace(MUSIC_NOUN_HEAD_RE, '')
    88→  }
    89→  q = q.replace(/^[\s,.:;!?—-]+|[\s,.:;!?—-]+$/g, '').trim()
    90→  if (!q || MUSIC_NOUN_ONLY_RE.test(q)) return ''
    91→  return q.slice(0, 120)
    92→}

   735→                  ? 'Остановить запись'
   736→                  : micDesired && (tts.speaking || music !== null)

   766→              />
   767→            )}
   768→
   769→            <button

   772→              aria-label="Дополнительно"
   773→              title="Дополнительно"
