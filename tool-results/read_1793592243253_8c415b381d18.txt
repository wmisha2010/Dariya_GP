    82→    82→function extractMusicQuery(raw: string): string {
    83→    83→  let q = raw.replace(MUSIC_VERB_RE, ' ').replace(MUSIC_FILLER_RE, ' ')
    84→    84→  let prev = ''
    85→    85→  while (prev !== q) {
    86→    86→    prev = q
    87→    87→    q = q.replace(MUSIC_NOUN_HEAD_RE, '')
    88→    88→  }
    89→    89→  q = q.replace(/^[\s,.:;!?—-]+|[\s,.:;!?—-]+$/g, '').trim()
    90→    90→  if (!q || MUSIC_NOUN_ONLY_RE.test(q)) return ''
    91→    91→  return q.slice(0, 120)
    92→    92→}
