    62→export default function Home() {
    63→  const { toast } = useToast()
    64→
    65→  // Данные
    66→  const [characters, setCharacters] = useState<Character[]>([])
    67→  const [activeCharacter, setActiveCharacter] = useState<Character | null>(null)
    68→  const [messages, setMessages] = useState<ChatMessage[]>([])
    69→  const [streamingText, setStreamingText] = useState('')
    70→
    71→  // Настройки
    72→  const [searchOn, setSearchOn] = useState(false)
    73→  const [voiceOn, setVoiceOn] = useState(true)
    74→  const [memoryOpen, setMemoryOpen] = useState(false)
    75→  /** Сейчас играет трек (YouTube-плеер смонтирован) */
    76→  const [music, setMusic] = useState<{ videoId: string; title: string; url: string } | null>(null)
    77→
    78→  // Ввод
    79→  const [inputValue, setInputValue] = useState('')
    80→  const [interim, setInterim] = useState('')
    81→  const [micListening, setMicListening] = useState(false)
    82→  const [busy, setBusy] = useState(false)
    83→
    84→  const sendingRef = useRef(false)
    85→  const voiceOnRef = useRef(voiceOn)
    86→  voiceOnRef.current = voiceOn
    87→  const searchOnRef = useRef(searchOn)
    88→  searchOnRef.current = searchOn
    89→  const textareaRef = useRef<HTMLTextAreaElement>(null)
    90→
    91→  // ─── Эхо-защита: микрофон не должен слышать голос самого ассистента ─────
    92→  const [micDesired, setMicDesired] = useState(false) // пользователь включил микрофон
    93→  const micDesiredRef = useRef(false)
    94→  const ttsSpeakingRef = useRef(false)
    95→  const musicPlayingRef = useRef(false)
    96→  const wasSpeakingRef = useRef(false)
    97→  const echoMuteUntilRef = useRef(0)
    98→  const resumeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    99→  const lastBusyToastRef = useRef(0)
   100→
   101→  // ─── Озвучка: серверный нейроголос Edge (голос выбирается кнопкой «Голос») ─
   102→  const [ttsVoice, setTtsVoice] = useState<string>('svetlana')
   103→
   104→  // Восстанавливаем выбранный голос с прошлого визита
   105→  useEffect(() => {
   106→    const saved = window.localStorage.getItem(TTS_VOICE_STORAGE_KEY)
   107→    if (saved && TTS_VOICES.some((v) => v.id === saved)) setTtsVoice(saved)
   108→  }, [])
   109→
   110→  /** Циклическое переключение голоса с сохранением выбора */
   111→  const cycleVoice = useCallback(() => {
   112→    setTtsVoice((prev) => {
   113→      const idx = TTS_VOICES.findIndex((v) => v.id === prev)
   114→      const next = TTS_VOICES[(idx + 1) % TTS_VOICES.length]
   115→      window.localStorage.setItem(TTS_VOICE_STORAGE_KEY, next.id)
   116→      return next.id
   117→    })
   118→  }, [])
   119→
   120→  const currentVoiceLabel =
   121→    TTS_VOICES.find((v) => v.id === ttsVoice)?.name ?? TTS_VOICES[0].name
   122→
   123→  /** Браузер заблокировал автозапуск звука — объясняем, как включить */
   124→  const handlePlaybackBlocked = useCallback(() => {
   125→    toast({
   126→      title: 'Браузер заблокировал звук',
   127→      description: 'Коснись экрана в любом месте — и я продолжу говорить голосом.',
   128→    })
   129→  }, [toast])
   130→
   131→  const tts = useTtsQueue({ voiceId: ttsVoice, onPlaybackBlocked: handlePlaybackBlocked })
   132→
   133→  ttsSpeakingRef.current = tts.speaking
   134→  musicPlayingRef.current = music !== null
   135→  /** Всё услышанное во время озвучки/музыки и сразу после — эхо, а не речь пользователя */
   136→  const isEcho = () =>
   137→    ttsSpeakingRef.current || musicPlayingRef.current || Date.now() < echoMuteUntilRef.current
   138→
   139→  /** Локальный ответ без LLM: музыкальные команды отвечают мгновенно */
   140→  const assistantLocalReply = useCallback(
   141→    (content: string) => {
   142→      setMessages((prev) => [
   143→        ...prev,
   144→        { id: `assistant-${Date.now()}`, role: 'assistant', content, createdAt: new Date().toISOString() },
   145→      ])
   146→      if (voiceOnRef.current) tts.speak(content)
   147→    },
   148→    [tts]
   149→  )
   150→
   151→  // ─── Отправка сообщения ─────────────────────────────────────────────────
   152→  const send = useCallback(
   153→    async (rawText: string) => {
   154→      const text = rawText.trim()
   155→      if (!text || sendingRef.current || !activeCharacter) return
   156→      sendingRef.current = true
   157→      setBusy(true)
   158→      setInputValue('')
   159→      setInterim('')
   160→      tts.stop()
   161→
   162→      const now = new Date().toISOString()
   163→      const localId = `local-${Date.now()}`
   164→      setMessages((prev) => [
   165→        ...prev,
   166→        { id: localId, role: 'user', content: text, createdAt: now },
   167→      ])
   168→
   169→      try {
   170→        // Голосовое управление музыкой: «выключи музыку» — гасим плеер без LLM
   171→        if (MUSIC_STOP_RE.test(text) && MUSIC_TOPIC_RE.test(text)) {
   172→          tts.stop()
   173→          setMusic(null)
   174→          assistantLocalReply('Хорошо, выключила музыку 🎧')
   175→          return
   176→        }
   177→
   178→        // Голосовое управление музыкой: «включи песню…» — ищем и играем
   179→        if (MUSIC_TRIGGER_RE.test(text) && MUSIC_TOPIC_RE.test(text)) {
   180→          const query =
   181→            text
   182→              .replace(MUSIC_TRIGGER_RE, '')
   183→              .replace(/\s*пожалуйста\s*/gi, ' ')
   184→              .trim()
   185→              .slice(0, 120) || 'музыка'
   186→          const res = await fetch('/api/music', {
   187→            method: 'POST',
   188→            headers: { 'Content-Type': 'application/json' },
   189→            body: JSON.stringify({ query }),
   190→          })
   191→          if (!res.ok) throw new Error('nothing found')
   192→          const json = (await res.json()) as { track?: { videoId: string; title: string; url: string } }
   193→          if (!json.track) throw new Error('nothing found')
   194→          tts.stop()
   195→          setMusic({ videoId: json.track.videoId, title: json.track.title, url: json.track.url })
   196→          assistantLocalReply(`Включаю «${json.track.title}» 🎵\n${json.track.url}`)
   197→          return
   198→        }
   199→
   200→        // Запрос к ассистенту с автоповтором: перегрузка API (429/503) обычно
   201→        // проходит за секунды — молча ждём и пробуем снова; красная ошибка —
   202→        // только если не помогло даже после трёх попыток
   203→        let res: Response | null = null
   204→        let lastErrorText = ''
   205→        for (let attempt = 0; attempt < 3; attempt++) {
   206→          const r = await fetch('/api/chat', {
   207→            method: 'POST',
   208→            headers: { 'Content-Type': 'application/json' },
   209→            body: JSON.stringify({
   210→              characterId: activeCharacter.id,
   211→              message: text,
   212→              search: searchOnRef.current,
   213→            }),
   214→          })
   215→          if (r.ok && r.body) {
   216→            res = r
   217→            break
   218→          }
   219→          const payload = (await r.json().catch(() => null)) as { error?: string } | null
   220→          lastErrorText = payload?.error ?? `Ошибка ${r.status}`
   221→          const retryable = r.status === 429 || r.status === 503 || r.status >= 500
   222→          if (!retryable || attempt === 2) break
   223→          if (attempt === 0) toast({ title: 'Модель притормозила — пробую ещё раз…' })
   224→          await new Promise((resolve) => setTimeout(resolve, 3500 * (attempt + 1)))
   225→        }
   226→        if (!res || !res.body) {
   227→          throw new Error(lastErrorText || 'Ассистент не смог ответить')
   228→        }
   229→
   230→        const reader = res.body.getReader()
   231→        const decoder = new TextDecoder()
   232→        let full = ''
   233→
   234→        while (true) {
   235→          const { done, value } = await reader.read()
   236→          if (done) break
   237→          const delta = decoder.decode(value, { stream: true })
   238→          if (!delta) continue
   239→          full += delta
   240→          setStreamingText(full)
   241→          if (voiceOnRef.current) tts.enqueue(delta)
   242→        }
   243→        if (voiceOnRef.current) tts.endStream()
   244→
   245→        setStreamingText('')
   246→        if (!full.trim()) {
   247→          // Пустой ответ — не тишина, а понятное сообщение
   248→          throw new Error('Ассистент не успел ответить — попробуй ещё раз')
   249→        }
   250→        setMessages((prev) => [
   251→          ...prev,
   252→          {
   253→            id: `assistant-${Date.now()}`,
   254→            role: 'assistant',
   255→            content: full.trim(),
   256→            createdAt: new Date().toISOString(),
   257→          },
   258→        ])
   259→      } catch (error) {
   260→        setStreamingText('')
   261→        toast({
   262→          title: 'Не получилось',
   263→          description: error instanceof Error ? error.message : 'Ассистент не смог ответить',
   264→          variant: 'destructive',
   265→        })
   266→      } finally {
   267→        sendingRef.current = false
   268→        setBusy(false)
   269→      }
   270→    },
   271→    [activeCharacter, assistantLocalReply, toast, tts]
   272→  )
   273→
   274→  // ─── Микрофон ───────────────────────────────────────────────────────────
   275→  const recognition = useSpeechRecognition({
   276→    onInterim: (text) => {
   277→      if (isEcho()) return
   278→      setInterim(text)
   279→    },
   280→    onFinal: (text) => {
   281→      if (isEcho()) return
   282→      const clean = text.trim()
   283→      if (!clean) return
   284→      if (sendingRef.current) {
   285→        // Ассистент ещё отвечает — не теряем молча, а подсказываем
   286→        if (Date.now() - lastBusyToastRef.current > 4000) {
   287→          lastBusyToastRef.current = Date.now()
   288→          toast({ title: 'Секунду…', description: 'Ассистент ещё отвечает — повтори сразу после' })
   289→        }
   290→        return
   291→      }
   292→      void send(clean)
   293→    },
   294→    onError: (code) => {
   295→      toast({
   296→        title: 'Микрофон недоступен',
   297→        description:
   298→          code === 'not-allowed' || code === 'service-not-allowed'
   299→            ? 'Разреши доступ к микрофону в настройках браузера'
   300→            : `Ошибка распознавания: ${code}`,
   301→        variant: 'destructive',
   302→      })
   303→    },
   304→  })
   305→
   306→  const setMicOn = useCallback((on: boolean) => {
   307→    micDesiredRef.current = on
   308→    setMicDesired(on)
   309→  }, [])
   310→
   311→  const toggleMic = useCallback(() => {
   312→    if (!recognition.supported) {
   313→      toast({
   314→        title: 'Голосовой ввод недоступен',
   315→        description: 'Открой приложение в Chrome или Edge — там работает распознавание речи',
   316→        variant: 'destructive',
   317→      })
   318→      return
   319→    }
   320→    if (recognition.listening) {
   321→      setMicOn(false)
   322→      recognition.stop()
   323→      setInterim('')
   324→      return
   325→    }
   326→    // Перебивание: глушим ассистента и музыку, слушаем
   327→    setMicOn(true)
   328→    tts.stop()
   329→    setMusic(null)
   330→    recognition.start()
   331→  }, [recognition, setMicOn, toast, tts])
   332→
   333→  useEffect(() => {
   334→    setMicListening(recognition.listening)
   335→  }, [recognition.listening])
   336→
   337→  // Пока звучит голос ассистента или музыка — приглушаем микрофон (он не слышит
   338→  // сам себя), после тишины возвращаем с паузой на затухание звука
   339→  const audible = tts.speaking || music !== null
   340→  useEffect(() => {
   341→    if (audible && !wasSpeakingRef.current) {
   342→      wasSpeakingRef.current = true
   343→      if (resumeTimerRef.current) {
   344→        clearTimeout(resumeTimerRef.current)
   345→        resumeTimerRef.current = null
   346→      }
   347→      recognition.pause()
   348→    } else if (!audible && wasSpeakingRef.current) {
   349→      wasSpeakingRef.current = false
   350→      echoMuteUntilRef.current = Date.now() + ECHO_GRACE_MS
   351→      if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current)
   352→      resumeTimerRef.current = setTimeout(() => {
   353→        resumeTimerRef.current = null
   354→        if (micDesiredRef.current) recognition.resume()
   355→      }, ECHO_GRACE_MS)
   356→    }
   357→  }, [audible, recognition.pause, recognition.resume])
   358→
   359→  useEffect(
   360→    () => () => {
   361→      if (resumeTimerRef.current) clearTimeout(resumeTimerRef.current)
   362→    },
   363→    []
   364→  )
   365→
   366→  // ─── Состояние ассистента для орбиса ────────────────────────────────────
   367→  const assistantState: AssistantState = micListening
   368→    ? 'listening'
   369→    : tts.speaking || music !== null
   370→      ? 'speaking'
   371→      : busy
   372→        ? 'thinking'
   373→        : 'idle'
   374→
   375→  // ─── Загрузка персонажей и истории ──────────────────────────────────────
   376→  const loadHistory = useCallback(async (characterId: string) => {
   377→    try {
   378→      const res = await fetch(`/api/messages?characterId=${encodeURIComponent(characterId)}`)
   379→      const json = await res.json()
   380→      setMessages(json.messages ?? [])
   381→    } catch {
   382→      setMessages([])
   383→    }
   384→  }, [])
   385→
   386→  useEffect(() => {
   387→    let cancelled = false
   388→    ;(async () => {
   389→      try {
   390→        const res = await fetch('/api/characters')
   391→        const json = await res.json()
   392→        const list: Character[] = json.characters ?? []
   393→        if (cancelled) return
   394→        setCharacters(list)
   395→        const first = list.find((c) => c.isDefault) ?? list[0] ?? null
   396→        if (first) {
   397→          setActiveCharacter(first)
   398→          await loadHistory(first.id)
   399→        }
   400→      } catch {
   401→        if (!cancelled) {
   402→          toast({
   403→            title: 'Ошибка загрузки',
   404→            description: 'Не удалось загрузить персонажей',
   405→            variant: 'destructive',
   406→          })
   407→        }
   408→      }
   409→    })()
   410→    return () => {
   411→      cancelled = true
   412→    }
   413→  }, [])
   414→
   415→  // ─── Очистить чат ───────────────────────────────────────────────────────
   416→  const clearChat = useCallback(async () => {
   417→    if (!activeCharacter) return
   418→    tts.stop()
   419→    setMusic(null)
   420→    try {
   421→      await fetch(`/api/messages?characterId=${encodeURIComponent(activeCharacter.id)}`, {
   422→        method: 'DELETE',
   423→      })
   424→      setMessages([])
   425→      toast({ title: 'История очищена', description: `${activeCharacter.name} всё ещё помнит факты о тебе` })
   426→    } catch {
   427→      toast({ title: 'Не удалось очистить историю', variant: 'destructive' })
   428→    }
   429→  }, [activeCharacter, toast, tts])
   430→
   431→  // ─── Быстрые действия ───────────────────────────────────────────────────
   432→  const toggleVoice = () => {
   433→    setVoiceOn((prev) => {
   434→      if (prev) tts.stop()
   435→      return !prev
   436→    })
   437→  }
   438→
   439→  const accent = activeCharacter?.accent ?? '#2dd4bf'
   440→  const greeting = activeCharacter
   441→    ? `${activeCharacter.emoji} Я ${activeCharacter.name} — ${activeCharacter.tagline.toLowerCase()}`
   442→    : 'Загружаю ассистентов…'
   443→
   444→  const canSend = inputValue.trim().length > 0 && !busy
   445→
   446→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   447→    if (e.key === 'Enter' && !e.shiftKey) {
   448→      e.preventDefault()
   449→      if (canSend) void send(inputValue)
   450→    }
   451→  }
   452→
   453→  return (
   454→    <div className="flex h-dvh flex-col overflow-hidden bg-[#070b12] text-white">
   455→      {/* Фоновые свечения */}
   456→      <div className="pointer-events-none fixed inset-0 overflow-hidden" aria-hidden="true">
   457→        <div
   458→          className="absolute -top-40 left-1/2 h-[500px] w-[800px] -translate-x-1/2 rounded-full opacity-20 blur-[120px] transition-colors duration-700"
   459→          style={{ backgroundColor: accent }}
   460→        />
   461→        <div className="absolute bottom-0 right-0 h-[300px] w-[400px] rounded-full bg-teal-500/5 blur-[100px]" />
   462→      </div>
   463→
   464→      {/* Шапка */}
   465→      <header className="relative z-10 flex items-center justify-between gap-3 border-b border-white/5 px-4 py-3 sm:px-6">
   466→        <div className="flex items-center gap-3">
   467→          <div
   468→            className="grid size-10 place-items-center rounded-2xl border border-white/10 bg-white/5 backdrop-blur-md"
   469→            style={{ boxShadow: `0 0 20px -8px ${accent}` }}
   470→          >
   471→            <Sparkles className="size-5" style={{ color: accent }} />
   472→          </div>
   473→          <div className="leading-tight">
   474→            <h1 className="text-lg font-bold tracking-wide">ДАРИЯ</h1>
   475→            <p className="text-[11px] text-white/40">голосовой ИИ-ассистент</p>
   476→          </div>
   477→        </div>
   478→      </header>
   479→
   480→      {/* Основная область: диалог сверху, композер снизу */}
   481→      <main className="relative z-10 mx-auto flex min-h-0 w-full max-w-3xl flex-1 flex-col px-4 pb-4 pt-4 sm:px-6">
   482→        {/* Диалог — занимает верх и скроллится по мере заполнения */}
   483→        <div className="min-h-0 flex-1">
   484→          <ChatPanel
   485→            messages={messages}
   486→            streamingText={streamingText}
   487→            accent={accent}
   488→            characterName={activeCharacter?.name ?? 'Ассистент'}
   489→            emptyHint={`Начни разговор — ${activeCharacter?.name ?? 'ассистент'} услышит и запомнит важное о тебе`}
   490→          />
   491→        </div>
   492→
   493→        {/* Композер: плеер, голограмма, подсказки и строка ввода */}
   494→        <div className="shrink-0">
   495→          {/* Плеер: музыка играет даже во время переписки */}
   496→          {music && (
   497→            <div className="mb-2.5">
   498→              <MusicPlayer
   499→                videoId={music.videoId}
   500→                title={music.title}
   501→                url={music.url}
   502→                accent={accent}
   503→                onClose={() => setMusic(null)}
   504→              />
   505→            </div>
   506→          )}
   507→
   508→          {/* Голограмма: тап — активировать микрофон, Дария слушает */}
   509→          <div className="flex flex-col items-center justify-center gap-1.5 text-center sm:flex-row sm:gap-4 sm:text-left">
   510→            <button
   511→              onClick={toggleMic}
   512→              aria-label={micListening ? 'Остановить микрофон' : 'Активировать голосового помощника'}
   513→              title={micListening ? 'Я уже слушаю — нажми, чтобы остановить' : 'Нажми — я слушаю'}
   514→              className="grid shrink-0 place-items-center rounded-full outline-none transition-all duration-200 hover:bg-white/[0.04] focus-visible:ring-2 focus-visible:ring-white/20 active:scale-95"
   515→            >
   516→              <VoiceOrb state={assistantState} accent={accent} size={124} />
   517→            </button>
   518→
   519→            <div className="flex min-w-0 flex-col gap-1">
   520→              <p className="line-clamp-2 text-sm font-medium text-white/85 sm:truncate">
   521→                {music
   522→                  ? '♪ Музыка играет'
   523→                  : assistantState === 'idle'
   524→                    ? greeting
   525→                    : STATUS_TEXT[assistantState]}
   526→              </p>
   527→              {interim ? (
   528→                <p className="line-clamp-2 text-sm italic text-white/45">«{interim}»</p>
   529→              ) : (
   530→                <p className="line-clamp-2 text-xs text-white/30 sm:truncate">{STATUS_TEXT[assistantState]}</p>
   531→              )}
   532→              {/* Звук заблокирован autoplay-политикой — ждём первого касания */}
   533→              {tts.blocked && (
   534→                <p
   535→                  role="status"
   536→                  className="w-fit animate-pulse rounded-full border border-amber-400/40 bg-amber-400/10 px-3.5 py-1.5 text-xs font-medium text-amber-300"
   537→                >
   538→                  Коснись экрана — включу голос
   539→                </p>
   540→              )}
   541→            </div>
   542→          </div>
   543→
   544→          {/* Быстрые подсказки и переключатели — всегда над строкой ввода */}
   545→          <div className="mt-2.5 flex flex-wrap items-center justify-center gap-2">
   546→            {QUICK_PROMPTS.map((prompt) => (
   547→              <button
   548→                key={prompt}
   549→                onClick={() => void send(prompt)}
   550→                className="rounded-full border border-white/10 bg-white/5 px-3 py-1.5 text-[11px] text-white/60 transition-all hover:border-white/20 hover:bg-white/10 hover:text-white/90 sm:text-xs"
   551→              >
   552→                {prompt}
   553→              </button>
   554→            ))}
   555→            <ToggleChip
   556→              active={searchOn}
   557→              onClick={() => setSearchOn((v) => !v)}
   558→              icon={<Globe className="size-3.5" />}
   559→              label="Поиск в интернете"
   560→            />
   561→            <ToggleChip
   562→              active={voiceOn}
   563→              onClick={toggleVoice}
   564→              icon={voiceOn ? <Volume2 className="size-3.5" /> : <VolumeX className="size-3.5" />}
   565→              label="Озвучка ответов"
   566→              title={`Озвучка ответов · Голос: ${tts.voiceName}`}
   567→            />
   568→            <button
   569→              onClick={cycleVoice}
   570→              title={`Голос ассистента: ${tts.voiceName}. Нажми, чтобы переключить на следующий`}
   571→              aria-label="Сменить голос ассистента"
   572→              className="flex items-center gap-1.5 rounded-full border border-white/10 bg-white/[0.03] px-3 py-1.5 text-xs text-white/45 transition-all duration-200 hover:border-teal-400/40 hover:bg-teal-400/10 hover:text-teal-300"
   573→            >
   574→              <AudioLines className="size-3.5" />
   575→              <span className="hidden sm:inline">Голос: {currentVoiceLabel}</span>
   576→            </button>
   577→            <ToggleChip
   578→              active={memoryOpen}
   579→              onClick={() => setMemoryOpen(true)}
   580→              icon={<Brain className="size-3.5" />}
   581→              label="Память и дневник"
   582→            />
   583→            <ToggleChip
   584→              active={false}
   585→              onClick={() => void clearChat()}
   586→              icon={<Trash2 className="size-3.5" />}
   587→              label="Очистить чат"
   588→            />
   589→          </div>
   590→
   591→          {/* Строка ввода */}
   592→          <div className="mt-2.5 flex items-end gap-2 rounded-3xl border border-white/10 bg-white/[0.04] p-2 backdrop-blur-md">
   593→            <button
   594→              onClick={toggleMic}
   595→              className={cn(
   596→                'grid size-11 shrink-0 place-items-center rounded-full transition-all duration-200',
   597→                micListening
   598→                  ? 'animate-pulse bg-red-500/90 text-white shadow-[0_0_24px_-4px_rgba(239,68,68,0.7)]'
   599→                  : 'text-white/60 hover:bg-white/10 hover:text-white'
   600→              )}
   601→              style={
   602→                micListening
   603→                  ? undefined
   604→                  : { boxShadow: `inset 0 0 0 1px rgba(255,255,255,0.08)` }
   605→              }
   606→              aria-label={micListening ? 'Остановить микрофон' : 'Говорить'}
   607→              title={
   608→                micListening
   609→                  ? 'Остановить'
   610→                  : micDesired && (tts.speaking || music !== null)
   611→                    ? 'Играет музыка или мой голос — кликни, чтобы перебить'
   612→                    : 'Говорить голосом'
   613→              }
   614→            >
   615→              {micListening ? (
   616→                <Square className="size-4" />
   617→              ) : micDesired && (tts.speaking || music !== null) ? (
   618→                <MicOff className="size-5 opacity-50" />
   619→              ) : (
   620→                <Mic className="size-5" />
   621→              )}
   622→            </button>
   623→
   624→            <Textarea
   625→              ref={textareaRef}
   626→              value={inputValue}
   627→              onChange={(e) => setInputValue(e.target.value)}
   628→              onKeyDown={handleKeyDown}
   629→              placeholder={
   630→                recognition.supported
   631→                  ? 'Напиши или скажи голосом…'
   632→                  : 'Напиши сообщение… (голос работает в Chrome/Edge)'
   633→              }
   634→              rows={1}
   635→              className="max-h-32 min-h-11 flex-1 resize-none border-0 bg-transparent py-2.5 text-sm text-white placeholder:text-white/30 focus-visible:ring-0 focus-visible:ring-offset-0"
   636→            />
   637→
   638→            <Button
   639→              onClick={() => void send(inputValue)}
   640→              disabled={!canSend}
   641→              size="icon"
   642→              className="size-11 shrink-0 rounded-full text-black transition-opacity disabled:opacity-30"
   643→              style={{ backgroundColor: accent }}
   644→              aria-label="Отправить"
   645→            >
   646→              <SendHorizontal className="size-4" />
   647→            </Button>
   648→          </div>
   649→
   650→          <p className="mt-2 text-center text-[11px] text-white/25">
   651→            Enter — отправить · Shift+Enter — новая строка · Тап по голограмме — включить голос
   652→            <br />
   653→            Скажи «включи песню…» или «спокойную музыку» — найду и заиграю · Клик по микрофону во время ответа — перебить
   654→          </p>
   655→        </div>
   656→      </main>
   657→
   658→      <MemoryDialog open={memoryOpen} onOpenChange={setMemoryOpen} onMemoryChanged={() => {}} />
   659→    </div>
   660→  )
   661→}
   662→
   663→interface ToggleChipProps {
   664→  active: boolean
   665→  onClick: () => void

   667→  label: string
   668→  title?: string
   669→}
   670→
   671→function ToggleChip({ active, onClick, icon, label, title }: ToggleChipProps) {
   672→  return (
   673→    <button
   674→      onClick={onClick}
   675→      title={title}
   676→      className={cn(
   677→        'flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-xs transition-all duration-200',
   678→        active
   679→          ? 'border-teal-400/40 bg-teal-400/10 text-teal-300'
   680→          : 'border-white/10 bg-white/[0.03] text-white/45 hover:border-white/20 hover:text-white/75'
   681→      )}
   682→      aria-pressed={active}
   683→    >
   684→      {icon}
   685→      <span className="hidden sm:inline">{label}</span>
   686→    </button>
   687→  )
   688→}
